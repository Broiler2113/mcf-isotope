cmake_minimum_required(VERSION 3.22)
project(mcf_visuals LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GODOT_CPP_DIR "" CACHE PATH "Pinned godot-cpp source directory")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(mcf_visuals SHARED src/mesh_builder.cpp src/visibility_kernel.cpp
    src/pathfinding_kernel.cpp src/ai_kernel.cpp src/terrain_kernel.cpp)
target_link_libraries(mcf_visuals PRIVATE godot-cpp)
# Match GDScript's operation order. Fused or reassociated arithmetic changes vertices.
if(MSVC)
    target_compile_options(mcf_visuals PRIVATE /W4 /fp:strict)
    target_compile_options(godot-cpp PRIVATE /fp:strict)
else()
    target_compile_options(mcf_visuals PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
    target_compile_options(godot-cpp PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(WIN32)
    set(MCF_PLATFORM windows)
elseif(APPLE)
    set(MCF_PLATFORM macos)
else()
    set(MCF_PLATFORM linux)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(MCF_ARCH arm64)
else()
    set(MCF_ARCH x86_64)
endif()
set(MCF_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/../../addons/mcf_visuals/bin/${MCF_PLATFORM}-${MCF_ARCH}")
set_target_properties(mcf_visuals PROPERTIES PREFIX ""
    LIBRARY_OUTPUT_DIRECTORY "${MCF_OUTPUT}"
    RUNTIME_OUTPUT_DIRECTORY "${MCF_OUTPUT}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${MCF_OUTPUT}")
