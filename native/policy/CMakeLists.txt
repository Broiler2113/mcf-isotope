cmake_minimum_required(VERSION 3.22)
project(mcf_policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GODOT_CPP_DIR "" CACHE PATH "Pinned godot-cpp source directory")
set(ORT_ROOT "" CACHE PATH "ONNX Runtime CPU distribution directory")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
option(MCF_EXTENSION "Build the Godot extension in addition to the parity probe" ON)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ORT_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(ORT_LIBRARY onnxruntime PATHS "${ORT_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_library(mcf_core STATIC src/features.cpp src/runtime.cpp)
target_include_directories(mcf_core PUBLIC src "${ORT_INCLUDE}" "${JSON_INCLUDE_DIR}")
target_link_libraries(mcf_core PUBLIC "${ORT_LIBRARY}")
if(MSVC)
    target_compile_options(mcf_core PRIVATE /EHsc /fp:precise)
else()
    target_compile_options(mcf_core PRIVATE -Wall -Wextra -fno-fast-math)
endif()
add_executable(mcf_policy_probe src/probe.cpp)
target_link_libraries(mcf_policy_probe PRIVATE mcf_core)
set_target_properties(mcf_policy_probe PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib")
if(WIN32)
    add_custom_command(TARGET mcf_policy_probe POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:mcf_policy_probe>")
endif()
if(MCF_EXTENSION)
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
    set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
    add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
    add_library(mcf_policy SHARED src/mcf_policy.cpp)
    target_link_libraries(mcf_policy PRIVATE mcf_core godot-cpp)
    if(WIN32)
        set(MCF_PLATFORM windows)
    elseif(APPLE)
        set(MCF_PLATFORM macos)
    else()
        set(MCF_PLATFORM linux)
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
        set(MCF_ARCH arm64)
    else()
        set(MCF_ARCH x86_64)
    endif()
    set(MCF_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/../../addons/mcf_policy/bin/${MCF_PLATFORM}-${MCF_ARCH}")
    set_target_properties(mcf_policy PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${MCF_OUTPUT}"
        RUNTIME_OUTPUT_DIRECTORY "${MCF_OUTPUT}"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${MCF_OUTPUT}"
        PREFIX "" BUILD_RPATH "$ORIGIN;@loader_path"
        INSTALL_RPATH "$ORIGIN;@loader_path" BUILD_WITH_INSTALL_RPATH ON)
    # Keep the runtime's actual SONAME (Linux .so.1, macOS .1.dylib).
    if(WIN32)
        set(ORT_SHARED "${ORT_ROOT}/lib/onnxruntime.dll")
    elseif(APPLE)
        set(ORT_SHARED "${ORT_ROOT}/lib/libonnxruntime.1.22.0.dylib")
    else()
        set(ORT_SHARED "${ORT_ROOT}/lib/libonnxruntime.so.1.22.0")
    endif()
    if(APPLE)
        set(ORT_DEST libonnxruntime.1.22.0.dylib)
    elseif(WIN32)
        set(ORT_DEST onnxruntime.dll)
    else()
        set(ORT_DEST libonnxruntime.so.1)
    endif()
    add_custom_command(TARGET mcf_policy POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_SHARED}" "${MCF_OUTPUT}/${ORT_DEST}")
endif()
